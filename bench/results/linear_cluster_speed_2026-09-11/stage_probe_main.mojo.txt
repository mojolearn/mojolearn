# Lane linear-cluster-speed stage probe (2026-09-11). Not a check and not a
# timing row: it splits OUR IDENTICAL OLS, PCA and k-means fits into stages
# with a synchronize at every boundary, on the prepped benchmark blocks.
#
#   pixi run mojo build -I . -D MOJOLEARN_NUMERIC_IDENTICAL=1 stage_probe_main.mojo -o stage_probe
#   ./stage_probe <ols|pca|kmeans> <prefix> <rows> <cols> <reps>
#
# <prefix>_X.bin (float32 rows x cols), <prefix>_Xc.bin and <prefix>_yc.bin
# (OLS: the Python layer's centered design and target), <prefix>_init.bin
# (k-means: 64 x cols).
from std.memory import memcpy
from std.sys import argv
from std.time import perf_counter_ns
from std.collections.string import atol
from max.gpu.host import DeviceBuffer, DeviceContext

from cluster.estimator import kmeans_fit, plan_sum_scale
from cluster.impl.kmeans_params import INIT_ARRAY, METRIC_L2_EXPANDED
from core.column_stats import (
    STATS_TPB,
    diagonal_to_vector_kernel,
    xty_kernel,
)
from core.gemm import gemm_tn
from decomposition.checks.jacobi_eigh_device import (
    JACOBI_SWEEPS,
    JACOBI_TOL,
    JACOBI_TPB,
    jacobi_eigh_kernel,
)
from decomposition.impl.linalg.detail.pca import compute_covariance
from glm.estimator import ols_fit_host
from glm.impl.linalg.detail.lstsq import ols_equilibration_scale
from glm.impl.matrix.math import (
    matrix_vector_binary_mult_kernel,
    row_vector_binary_mult_kernel,
    vector_binary_mult_kernel,
)


def _load(ctx: DeviceContext, path: String, n: Int) raises -> List[UInt8]:
    var f = open(path, "r")
    var b = f.read_bytes()
    f.close()
    if len(b) != n * 4:
        raise Error(path + " holds " + String(len(b)) + " bytes, expected " + String(n * 4))
    return b^


def _ms(t0: Int) -> Float64:
    return Float64(Int(perf_counter_ns()) - t0) / 1.0e6


def _now() -> Int:
    return Int(perf_counter_ns())


def _jacobi(ctx: DeviceContext, mut a: DeviceBuffer[DType.float32], n: Int) raises -> Float64:
    """Launch the shipped device Jacobi on `a` (consumed); print sweeps and ms."""
    var q = ctx.enqueue_create_buffer[DType.float32](n * n)
    var info = ctx.enqueue_create_buffer[DType.float32](3)
    ctx.synchronize()
    var t0 = _now()
    ctx.enqueue_function[jacobi_eigh_kernel](
        a.unsafe_ptr(), q.unsafe_ptr(), info.unsafe_ptr(), Int32(n),
        Int32(JACOBI_SWEEPS), Float32(JACOBI_TOL),
        grid_dim=(1, 1, 1), block_dim=(JACOBI_TPB, 1, 1),
    )
    ctx.synchronize()
    var t = _ms(t0)
    var h = ctx.enqueue_create_host_buffer[DType.float32](3)
    ctx.enqueue_copy(dst_ptr=h.unsafe_ptr(), src_buf=info)
    ctx.synchronize()
    print("  jacobi n=" + String(n) + " ms=" + String(t) + " converged="
          + String(h.unsafe_ptr().unsafe_load(0)) + " rel=" + String(h.unsafe_ptr().unsafe_load(1))
          + " sweeps=" + String(h.unsafe_ptr().unsafe_load(2)))
    _ = q^
    _ = h^
    _ = info^
    return t


def main() raises:
    var args = argv()
    var mode = String(args[1])
    var prefix = String(args[2])
    var rows = Int(atol(String(args[3])))
    var cols = Int(atol(String(args[4])))
    var reps = Int(atol(String(args[5])))
    var cells = rows * cols
    var ctx = DeviceContext()
    var t0 = _now()
    if mode == "kmeans":
        var bx = _load(ctx, prefix + "_X.bin", cells)
        var bi = _load(ctx, prefix + "_init.bin", 64 * cols)
        var hx = ctx.enqueue_create_host_buffer[DType.float32](cells)
        var hi = ctx.enqueue_create_host_buffer[DType.float32](64 * cols)
        var hc = ctx.enqueue_create_host_buffer[DType.float32](64 * cols)
        var hl = ctx.enqueue_create_host_buffer[DType.uint32](rows)
        ctx.synchronize()
        memcpy(dest=hx.unsafe_ptr().bitcast[UInt8](), src=bx.unsafe_ptr(), count=cells * 4)
        memcpy(dest=hi.unsafe_ptr().bitcast[UInt8](), src=bi.unsafe_ptr(), count=64 * cols * 4)
        _ = bx^
        _ = bi^
        print("kmeans load ms=" + String(_ms(t0)))
        for rep in range(reps):
            t0 = _now()
            var s = plan_sum_scale(hx.unsafe_ptr(), rows, cols)
            var t_plan = _ms(t0)
            for i in range(64 * cols):
                hc.unsafe_ptr().unsafe_store(i, hi.unsafe_ptr().unsafe_load(i))
            t0 = _now()
            var res = kmeans_fit(
                ctx, hx.unsafe_ptr(), rows, cols, 64, hc.unsafe_ptr(), hl.unsafe_ptr(),
                hx.unsafe_ptr(), 0, max_iter=20, tol=1.0e-7, init=INIT_ARRAY,
                metric=METRIC_L2_EXPANDED, requested_sum_scale=s,
            )
            var t_fit = _ms(t0)
            print("kmeans rep=" + String(rep) + " plan_sum_scale_ms=" + String(t_plan)
                  + " fit_given_scale_ms=" + String(t_fit) + " n_iter=" + String(res.n_iter)
                  + " inertia=" + String(res.inertia) + " scale=" + String(s))
        _ = hx^
        _ = hi^
        _ = hc^
        _ = hl^
        return
    if mode == "pca":
        var bx = _load(ctx, prefix + "_X.bin", cells)
        var hx = ctx.enqueue_create_host_buffer[DType.float32](cells)
        ctx.synchronize()
        memcpy(dest=hx.unsafe_ptr().bitcast[UInt8](), src=bx.unsafe_ptr(), count=cells * 4)
        _ = bx^
        for rep in range(reps):
            t0 = _now()
            var x = ctx.enqueue_create_buffer[DType.float32](cells)
            var xa = ctx.enqueue_create_buffer[DType.float32](cells)
            var xa2 = ctx.enqueue_create_buffer[DType.float32](cells)
            var mu = ctx.enqueue_create_buffer[DType.float32](cols)
            var cov = ctx.enqueue_create_buffer[DType.float32](cols * cols)
            ctx.synchronize()
            var t_alloc = _ms(t0)
            t0 = _now()
            ctx.enqueue_copy(dst_buf=x, src_ptr=hx.unsafe_ptr())
            ctx.synchronize()
            var t_up = _ms(t0)
            t0 = _now()
            compute_covariance(ctx, x, xa, xa2, mu, cov, rows, cols, True)
            ctx.synchronize()
            var t_cov = _ms(t0)
            print("pca rep=" + String(rep) + " alloc_ms=" + String(t_alloc) + " upload_ms="
                  + String(t_up) + " covariance_ms=" + String(t_cov))
            _ = _jacobi(ctx, cov, cols)
            _ = x^
            _ = xa^
            _ = xa2^
            _ = mu^
        _ = hx^
        return
    if mode == "ols":
        var bx = _load(ctx, prefix + "_Xc.bin", cells)
        var by = _load(ctx, prefix + "_yc.bin", rows)
        var hx = ctx.enqueue_create_host_buffer[DType.float32](cells)
        var hy = ctx.enqueue_create_host_buffer[DType.float32](rows)
        var hw = ctx.enqueue_create_host_buffer[DType.float32](cols)
        ctx.synchronize()
        memcpy(dest=hx.unsafe_ptr().bitcast[UInt8](), src=bx.unsafe_ptr(), count=cells * 4)
        memcpy(dest=hy.unsafe_ptr().bitcast[UInt8](), src=by.unsafe_ptr(), count=rows * 4)
        _ = bx^
        _ = by^
        for rep in range(reps):
            t0 = _now()
            ols_fit_host(ctx, hx.unsafe_ptr(), hy.unsafe_ptr(), hw.unsafe_ptr(), rows, cols)
            var t_total = _ms(t0)
            t0 = _now()
            var x = ctx.enqueue_create_buffer[DType.float32](cells)
            var y = ctx.enqueue_create_buffer[DType.float32](rows)
            var xa = ctx.enqueue_create_buffer[DType.float32](cells)
            var xa2 = ctx.enqueue_create_buffer[DType.float32](cells)
            var cov = ctx.enqueue_create_buffer[DType.float32](cols * cols)
            var ab = ctx.enqueue_create_buffer[DType.float32](cols)
            var sv = ctx.enqueue_create_buffer[DType.float32](cols)
            var scale = ctx.enqueue_create_buffer[DType.float32](cols)
            ctx.synchronize()
            var t_alloc = _ms(t0)
            t0 = _now()
            ctx.enqueue_copy(dst_buf=x, src_ptr=hx.unsafe_ptr())
            ctx.enqueue_copy(dst_buf=y, src_ptr=hy.unsafe_ptr())
            ctx.synchronize()
            var t_up = _ms(t0)
            t0 = _now()
            gemm_tn(ctx, cov, x, xa, xa2, cols, cols, rows)
            ctx.synchronize()
            var t_gram = _ms(t0)
            t0 = _now()
            ctx.enqueue_function[xty_kernel](
                ab.unsafe_ptr(), x.unsafe_ptr(), y.unsafe_ptr(), Int32(rows), Int32(cols),
                grid_dim=(cols, 1, 1), block_dim=(STATS_TPB, 1, 1),
            )
            ctx.synchronize()
            var t_xty = _ms(t0)
            t0 = _now()
            ctx.enqueue_function[diagonal_to_vector_kernel](
                sv.unsafe_ptr(), cov.unsafe_ptr(), Int32(cols),
                grid_dim=((cols + 255) // 256, 1, 1), block_dim=(256, 1, 1),
            )
            var hd = ctx.enqueue_create_host_buffer[DType.float32](cols)
            var hs = ctx.enqueue_create_host_buffer[DType.float32](cols)
            ctx.enqueue_copy(dst_ptr=hd.unsafe_ptr(), src_buf=sv)
            ctx.synchronize()
            for i in range(cols):
                hs.unsafe_ptr().unsafe_store(i, ols_equilibration_scale(hd.unsafe_ptr().unsafe_load(i)))
            ctx.enqueue_copy(dst_buf=scale, src_ptr=hs.unsafe_ptr())
            ctx.synchronize()
            ctx.enqueue_function[row_vector_binary_mult_kernel](
                cov.unsafe_ptr(), scale.unsafe_ptr(), Int32(cols), Int32(cols),
                grid_dim=((cols * cols + 255) // 256, 1, 1), block_dim=(256, 1, 1),
            )
            ctx.enqueue_function[matrix_vector_binary_mult_kernel](
                cov.unsafe_ptr(), scale.unsafe_ptr(), Int32(cols), Int32(cols),
                grid_dim=((cols * cols + 255) // 256, 1, 1), block_dim=(256, 1, 1),
            )
            ctx.synchronize()
            var t_eq = _ms(t0)
            print("ols rep=" + String(rep) + " ols_fit_host_total_ms=" + String(t_total)
                  + " alloc_ms=" + String(t_alloc) + " upload_ms=" + String(t_up)
                  + " gram_ms=" + String(t_gram) + " xty_ms=" + String(t_xty)
                  + " equilibrate_ms=" + String(t_eq) + " coef0=" + String(hw.unsafe_ptr().unsafe_load(0)))
            _ = _jacobi(ctx, cov, cols)
            _ = hd^
            _ = hs^
            _ = x^
            _ = y^
            _ = xa^
            _ = xa2^
            _ = ab^
            _ = sv^
            _ = scale^
        _ = hx^
        _ = hy^
        _ = hw^
        return
    raise Error("mode must be ols, pca or kmeans")
